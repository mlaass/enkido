# nkido CLI - Unified synth tool with SDL2 audio output

find_package(SDL2 REQUIRED)

# Bytecode disassembly formatter. Lifted into its own static library so
# akkado/tests/test_bytecode_snapshot.cpp can link the formatter without
# pulling in the rest of the CLI binary.
add_library(nkido_bytecode_dump STATIC
    bytecode_dump.cpp
)

target_include_directories(nkido_bytecode_dump
    PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
)

target_link_libraries(nkido_bytecode_dump
    PUBLIC
        cedar
    PRIVATE
        nkido_compiler_options
)

# Target label is `nkido_cli` to avoid colliding with potential future
# library/alias targets named `nkido`. OUTPUT_NAME controls the produced
# binary filename ("nkido" / "nkido.exe").
add_executable(nkido_cli
    main.cpp
    audio_engine.cpp
    asset_loader.cpp
    bytecode_loader.cpp
    midi_input.cpp
    program_loader.cpp
    serve_mode.cpp
    ui/bitmap_font.cpp
    ui/text_buffer.cpp
    ui/ui_mode.cpp
)

set_target_properties(nkido_cli PROPERTIES
    OUTPUT_NAME nkido
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
)

target_link_libraries(nkido_cli
    PRIVATE
        akkado
        cedar
        SDL2::SDL2
        rtmidi
        nkido_bytecode_dump
        nkido_compiler_options
)

target_include_directories(nkido_cli
    PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}
)

# Built-in default sample kit. The compile-time path makes in-tree builds
# resolve `s"bd"` etc. without setup; installed builds fall back to the
# `share/nkido/default_kit/strudel.json` rule below.
target_compile_definitions(nkido_cli PRIVATE
    NKIDO_DEFAULT_KIT_PATH="${CMAKE_SOURCE_DIR}/web/static/samples/bpb_808_clean/strudel.json"
    NKIDO_SOUNDFONT_PATH="${CMAKE_SOURCE_DIR}/web/static/soundfonts"
    NKIDO_TDM_CATALOG_PATH="${CMAKE_SOURCE_DIR}/web/static/samples/tidal-drum-machines/catalog.json"
)

# Platform-specific: SDL2main needed on Windows
if(WIN32)
    target_link_libraries(nkido_cli PRIVATE SDL2::SDL2main)
endif()

install(TARGETS nkido_cli
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)

install(DIRECTORY ${CMAKE_SOURCE_DIR}/web/static/samples/bpb_808_clean/
    DESTINATION share/nkido/default_kit
    FILES_MATCHING
        PATTERN "*.wav"
        PATTERN "strudel.json"
)

# Tidal Drum Machines catalog (generated by scripts/generate-tdm-catalog.sh).
# Text only — the audio streams from GitHub on demand. `find_tdm_catalog_uri()`
# searches this install location so `.bank("RolandTR808")` auto-resolves.
install(FILES ${CMAKE_SOURCE_DIR}/web/static/samples/tidal-drum-machines/catalog.json
    DESTINATION share/nkido/tidal-drum-machines
)

# Bundled SoundFonts (matches the web app's DEFAULT_SOUNDFONTS table). The
# `resolve_soundfont_alias()` helper in asset_loader.cpp searches this
# directory at runtime so `soundfont(@, "gm", 0)` works without --soundfont.
install(DIRECTORY ${CMAKE_SOURCE_DIR}/web/static/soundfonts/
    DESTINATION share/nkido/soundfonts
    FILES_MATCHING
        PATTERN "*.sf2"
        PATTERN "*.sf3"
        PATTERN "LICENSE.md"
)

if(NKIDO_BUILD_TESTS)
    add_subdirectory(tests)
endif()
